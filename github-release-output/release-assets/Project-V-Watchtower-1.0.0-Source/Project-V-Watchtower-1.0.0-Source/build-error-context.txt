  180:       else if (character === '"') quoted = false;
  181:       else field += character;
  182:     } else if (character === '"') quoted = true;
  183:     else if (character === ',') { row.push(field); field = ''; }
  184:     else if (character === '\n') { row.push(field.replace(/\r$/, '')); rows.push(row); row = []; field = ''; }
  185:     else field += character;
  186:   }
  187:   row.push(field.replace(/\r$/, ''));
  188:   if (row.length > 1 || row[0]) rows.push(row);
  189:   return { id: `sheet-${crypto.randomUUID?.() ?? Math.random().toString(36).slice(2)}`, name, rows: rows.length ? rows : [['']], formulas: {} };
  190: }

  192: function crc32(bytes: Uint8Array): number {
  193:   let crc = 0xffffffff;
  194:   for (const byte of bytes) {
  195:     crc ^= byte;
  196:     for (let bit = 0; bit < 8; bit += 1) crc = (crc >>> 1) ^ (0xedb88320 & -(crc & 1));
  197:   }
  198:   return (crc ^ 0xffffffff) >>> 0;
  199: }

  201: function concat(parts: Uint8Array[]): Uint8Array {
  202:   const length = parts.reduce((sum, part) => sum + part.length, 0);
  203:   const output = new Uint8Array(length);
  204:   let offset = 0;
  205:   for (const part of parts) { output.set(part, offset); offset += part.length; }
  206:   return output;
  207: }

  209: function zipStore(entries: Array<{ name: string; data: Uint8Array }>): Uint8Array {
  210:   const localParts: Uint8Array[] = [];
  211:   const centralParts: Uint8Array[] = [];
  212:   let offset = 0;
  213:   for (const entry of entries) {
  214:     const name = textEncoder.encode(entry.name);
  215:     const crc = crc32(entry.data);
  216:     const local = new Uint8Array(30 + name.length + entry.data.length);
  217:     const localView = new DataView(local.buffer);
  218:     localView.setUint32(0, 0x04034b50, true);
  219:     localView.setUint16(4, 20, true);
  220:     localView.setUint16(6, 0x0800, true);
  221:     localView.setUint16(8, 0, true);
  222:     localView.setUint32(14, crc, true);
  223:     localView.setUint32(18, entry.data.length, true);
  224:     localView.setUint32(22, entry.data.length, true);
  225:     localView.setUint16(26, name.length, true);
  226:     local.set(name, 30);
  227:     local.set(entry.data, 30 + name.length);
  228:     localParts.push(local);
  229: 
  230:     const central = new Uint8Array(46 + name.length);
  231:     const centralView = new DataView(central.buffer);
  232:     centralView.setUint32(0, 0x02014b50, true);
  233:     centralView.setUint16(4, 20, true);
  234:     centralView.setUint16(6, 20, true);
  235:     centralView.setUint16(8, 0x0800, true);
  236:     centralView.setUint16(10, 0, true);
  237:     centralView.setUint32(16, crc, true);
  238:     centralView.setUint32(20, entry.data.length, true);
  239:     centralView.setUint32(24, entry.data.length, true);
  240:     centralView.setUint16(28, name.length, true);
  241:     centralView.setUint32(42, offset, true);
  242:     central.set(name, 46);
  243:     centralParts.push(central);
  244:     offset += local.length;
  245:   }
  246:   const central = concat(centralParts);
  247:   const eocd = new Uint8Array(22);
  248:   const eocdView = new DataView(eocd.buffer);
  249:   eocdView.setUint32(0, 0x06054b50, true);
  250:   eocdView.setUint16(8, entries.length, true);
  251:   eocdView.setUint16(10, entries.length, true);
  252:   eocdView.setUint32(12, central.length, true);
  253:   eocdView.setUint32(16, offset, true);
  254:   return concat([...localParts, central, eocd]);
  255: }

  262: function worksheetXml(sheet: LiteSheet): string {
  263:   const rows = sheet.rows.map((row, rowIndexValue) => {
  264:     const cells = row.map((value, column) => {
  265:       const reference = cellReference(rowIndexValue, column);
  266:       const formula = sheet.formulas[reference];
  267:       if (formula) return `<c r="${reference}"><f>${xmlEscape(formula)}</f><v>${xmlEscape(value)}</v></c>`;
  268:       const trimmed = value.trim();
  269:       if (trimmed !== '' && /^-?(?:\d+\.?\d*|\.\d+)(?:[eE][+-]?\d+)?$/.test(trimmed)) return `<c r="${reference}"><v>${trimmed}</v></c>`;
  270:       if (/^(TRUE|FALSE)$/i.test(trimmed)) return `<c r="${reference}" t="b"><v>${/^TRUE$/i.test(trimmed) ? '1' : '0'}</v></c>`;
  271:       return `<c r="${reference}" t="inlineStr"><is><t xml:space="preserve">${xmlEscape(value)}</t></is></c>`;
  272:     }).join('');
  273:     return `<row r="${rowIndexValue + 1}">${cells}</row>`;
  274:   }).join('');
